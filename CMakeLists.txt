cmake_minimum_required(VERSION 3.24)
project(TidyDesk VERSION 0.4.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "TidyDesk requires Windows")
endif()
add_executable(TidyDeskProbe probe.cpp)
add_executable(TidyDesk WIN32 main.cpp settings.cpp desktop_model.cpp window_rule.cpp app.rc)
add_library(TidyDeskTap SHARED tap.cpp)
add_executable(TidyDeskAttach WIN32 attach.cpp)
add_executable(TidyDeskStop WIN32 stop.cpp)
add_executable(TidyDeskDesktop WIN32 desktop.cpp desktop_model.cpp app.rc)
add_executable(TidyDeskIcons WIN32 arrows.cpp app.rc)
foreach(target TidyDeskDesktop TidyDeskIcons TidyDeskProbe TidyDesk TidyDeskTap TidyDeskAttach TidyDeskStop)
target_compile_features(${target} PRIVATE cxx_std_17)
target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/utf-8;/permissive-;/EHsc>)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
endforeach()
target_link_libraries(TidyDeskProbe PRIVATE user32 psapi)
target_link_libraries(TidyDesk PRIVATE user32 shell32 ole32 uuid imm32 comctl32 gdi32 advapi32 dwmapi uxtheme)
target_link_libraries(TidyDeskAttach PRIVATE user32 ole32)
target_link_libraries(TidyDeskStop PRIVATE user32)
target_link_libraries(TidyDeskTap PRIVATE user32 ole32 oleaut32 windowsapp)
enable_testing()
add_test(NAME cli-help COMMAND TidyDeskProbe --help)
add_test(NAME invalid-argument COMMAND TidyDeskProbe --invalid)
set_tests_properties(invalid-argument PROPERTIES WILL_FAIL TRUE)
add_test(NAME host-startup-self-test COMMAND TidyDesk --self-test)
add_test(NAME desktop-maximized-window-rule COMMAND TidyDesk --test-window-rule)

if(MSVC)
  target_link_options(TidyDesk PRIVATE /MANIFEST:NO)
endif()


target_link_libraries(TidyDeskDesktop PRIVATE user32 shell32 ole32 uuid comctl32 gdi32 advapi32)
target_link_libraries(TidyDeskIcons PRIVATE user32 shell32 ole32 advapi32)
add_test(NAME desktop-model-recovery COMMAND TidyDeskDesktop --test-model)
if(MSVC)
 target_link_options(TidyDeskDesktop PRIVATE /MANIFEST:NO)
 target_link_options(TidyDeskIcons PRIVATE /MANIFEST:NO)
endif()
