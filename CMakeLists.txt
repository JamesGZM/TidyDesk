cmake_minimum_required(VERSION 3.24)
project(LiteTaskbar VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "LiteTaskbar requires Windows")
endif()
add_executable(LiteTaskbarProbe probe.cpp)
target_compile_features(LiteTaskbarProbe PRIVATE cxx_std_17)
target_compile_definitions(LiteTaskbarProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(LiteTaskbarProbe PRIVATE user32 psapi)
if(MSVC)
  target_compile_options(LiteTaskbarProbe PRIVATE /W4 /WX /utf-8 /permissive-)
  set_property(TARGET LiteTaskbarProbe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
enable_testing()
add_test(NAME cli-help COMMAND LiteTaskbarProbe --help)
add_test(NAME invalid-argument COMMAND LiteTaskbarProbe --invalid)
set_tests_properties(invalid-argument PROPERTIES WILL_FAIL TRUE)
