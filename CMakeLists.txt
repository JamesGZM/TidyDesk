cmake_minimum_required(VERSION 3.24)
project(TidyDesk VERSION 0.4.9 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "TidyDesk requires Windows")
endif()
add_executable(TidyDeskProbe probe.cpp)
add_executable(TidyDesk WIN32 main.cpp settings.cpp desktop_model.cpp window_rule.cpp app.rc)
add_library(TidyDeskTap SHARED tap.cpp)
add_executable(TidyDeskAttach WIN32 attach.cpp)
add_executable(TidyDeskStop WIN32 stop.cpp)
add_executable(TidyDeskDesktop WIN32 desktop.cpp desktop_model.cpp app.rc)
add_executable(TidyDeskIcons WIN32 arrows.cpp app.rc)
add_executable(TidyDeskTests tests.cpp desktop_model.cpp)
foreach(target TidyDeskTests TidyDeskDesktop TidyDeskIcons TidyDeskProbe TidyDesk TidyDeskTap TidyDeskAttach TidyDeskStop)
target_compile_features(${target} PRIVATE cxx_std_17)
target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/utf-8;/permissive-;/EHsc>)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
endforeach()
target_link_libraries(TidyDeskProbe PRIVATE user32 psapi)
target_link_libraries(TidyDesk PRIVATE user32 shell32 ole32 uuid imm32 comctl32 gdi32 advapi32 dwmapi uxtheme gdiplus)
target_link_libraries(TidyDeskAttach PRIVATE user32 ole32)
target_link_libraries(TidyDeskStop PRIVATE user32)
target_link_libraries(TidyDeskTap PRIVATE user32 ole32 oleaut32 windowsapp)
enable_testing()
add_test(NAME cli-help COMMAND TidyDeskProbe --help)
add_test(NAME invalid-argument COMMAND TidyDeskProbe --invalid)
set_tests_properties(invalid-argument PROPERTIES WILL_FAIL TRUE)
add_test(NAME host-startup-self-test COMMAND TidyDesk --self-test)
add_test(NAME desktop-maximized-window-rule COMMAND TidyDesk --test-window-rule)

if(MSVC)
  target_link_options(TidyDesk PRIVATE /MANIFEST:NO)
endif()


target_link_libraries(TidyDeskDesktop PRIVATE user32 shell32 ole32 uuid comctl32 comdlg32 gdi32 gdiplus advapi32)
target_link_libraries(TidyDeskIcons PRIVATE user32 shell32 ole32 uuid advapi32 gdi32)
add_test(NAME shortcut-overlay-rendering COMMAND TidyDeskIcons --test-overlay)
add_test(NAME desktop-model-recovery COMMAND TidyDeskDesktop --test-model)
add_test(NAME shortcut-arrow-backup COMMAND TidyDeskIcons --test-backup)
target_link_libraries(TidyDeskTests PRIVATE user32 shell32 ole32 uuid advapi32 comctl32)
add_test(NAME desktop-file-transfer COMMAND TidyDeskTests)
set_tests_properties(desktop-file-transfer PROPERTIES TIMEOUT 30)
if(MSVC)
 target_link_options(TidyDeskDesktop PRIVATE /MANIFEST:NO)
 target_link_options(TidyDeskIcons PRIVATE /MANIFEST:NO)
endif()

add_test(NAME desktop-grid-interaction COMMAND TidyDeskDesktop --test-geometry)
add_test(NAME desktop-visible-frame COMMAND TidyDeskDesktop --test-frame)
add_test(NAME desktop-icon-color-alpha COMMAND TidyDeskDesktop --test-icon-pixels)

add_test(NAME explicit-settings-apply COMMAND TidyDesk --test-settings)
set_tests_properties(explicit-settings-apply PROPERTIES TIMEOUT 30)
add_test(NAME shortcut-source-icon COMMAND TidyDeskDesktop --test-shortcut-icon)
add_test(NAME collection-folder-rename COMMAND TidyDeskDesktop --test-rename)
add_test(NAME permission-request-validation COMMAND TidyDeskIcons --test-permissions)

add_test(NAME retired-bootstrap-rejected COMMAND TidyDeskIcons --install-service)
set_tests_properties(retired-bootstrap-rejected PROPERTIES WILL_FAIL TRUE)
