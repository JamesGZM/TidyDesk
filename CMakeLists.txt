cmake_minimum_required(VERSION 3.24)
project(LiteTaskbar VERSION 0.2.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "LiteTaskbar requires Windows")
endif()
add_executable(LiteTaskbarProbe probe.cpp)
add_executable(LiteTaskbar WIN32 main.cpp)
add_library(LiteTaskbarTap SHARED tap.cpp)
foreach(target LiteTaskbarProbe LiteTaskbar LiteTaskbarTap)
target_compile_features(${target} PRIVATE cxx_std_17)
target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive- /EHsc)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
endforeach()
target_link_libraries(LiteTaskbarProbe PRIVATE user32 psapi)
target_link_libraries(LiteTaskbar PRIVATE user32 shell32 ole32)
target_link_libraries(LiteTaskbarTap PRIVATE user32 ole32 oleaut32 windowsapp)
enable_testing()
add_test(NAME cli-help COMMAND LiteTaskbarProbe --help)
add_test(NAME invalid-argument COMMAND LiteTaskbarProbe --invalid)
set_tests_properties(invalid-argument PROPERTIES WILL_FAIL TRUE)
add_test(NAME host-startup-self-test COMMAND LiteTaskbar --self-test)
